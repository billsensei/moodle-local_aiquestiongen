// Real-AI smoke test for local_aiquestiongen. Creates a course, a source article and a job, generates questions with
// the configured AI, regenerates one, approves everything and publishes to a new question bank and quiz.
// Run it on a TEST site only, after saving the AI settings: cp live-test.php.txt live-test.php; php live-test.php /path/to/moodle

define('CLI_SCRIPT', true);
if (empty($argv[1]) || !is_readable($argv[1] . '/config.php')) {
    fwrite(STDERR, "Usage: php live-test.php <moodle root folder>\n");
    exit(1);
}
require($argv[1] . '/config.php');
require_once($CFG->dirroot . '/course/lib.php');
use local_aiquestiongen\local\{job_manager, publisher, regenerator};
use local_aiquestiongen\ai\client_factory;
global $DB;
\core\session\manager::set_user(get_admin());
$admin = get_admin();
$article = "The Lighthouse Keeper's Cat\n\nOn a windy island off the coast of Scotland, an old lighthouse keeper named Angus lived with his grey cat, Misty. Every evening, Angus climbed the 120 steps to light the great lamp, and Misty always followed him. Last winter, a terrible storm hit the island. The waves were so high that the power failed, and the lamp went dark. Angus was worried because ships could not see the rocks. Misty, however, ran down the stairs and returned with a box of matches from the kitchen. Angus used them to light the old oil lamp, which was kept for emergencies. By morning, the storm had passed, and every ship had reached the harbour safely. The next week, the villagers brought a gift of fresh fish for Misty, and the local newspaper printed a photograph of the cat beside the lamp. Angus says he will retire next year, but he adds that Misty will never leave the lighthouse.";
$course = create_course((object) ['fullname' => 'Live AI test ' . date('His'), 'shortname' => 'live' . time(), 'category' => 1]);
echo "course {$course->id}\n";
$srcid = $DB->insert_record('local_aiquestiongen_source', (object) ['courseid' => $course->id, 'userid' => $admin->id,
    'title' => "The Lighthouse Keeper's Cat", 'text' => $article, 'sourcefileitemid' => 0, 'timecreated' => time()]);
$source = $DB->get_record('local_aiquestiongen_source', ['id' => $srcid]);
$jobid = job_manager::create_job($source, $admin->id, ['cefr' => 'B1', 'target' => 'junior high school students',
    'language' => 'English', 'count' => 3, 'tfcount' => 2, 'essaycount' => 1, 'clozecount' => 2, 'gapcount' => 1]);
echo "job $jobid queued\n";
$t0 = microtime(true);
$task = \core\task\manager::get_next_adhoc_task(time());
while ($task && (int) json_decode($task->get_custom_data_as_string())->jobid !== $jobid) {
    \core\task\manager::adhoc_task_complete($task);   // not ours (should not happen on this site)
    $task = \core\task\manager::get_next_adhoc_task(time());
}
$task->execute();
\core\task\manager::adhoc_task_complete($task);
printf("generation took %.1fs\n", microtime(true) - $t0);
$job = $DB->get_record('local_aiquestiongen_job', ['id' => $jobid]);
echo "status={$job->status} provider={$job->provider} tokens={$job->tokensused} error=" . ($job->error ?? '') . "\n";
$items = $DB->get_records('local_aiquestiongen_item', ['jobid' => $jobid], 'id');
$by = [];
foreach ($items as $i) {
    $by[$i->qtype . '/' . $i->status] = ($by[$i->qtype . '/' . $i->status] ?? 0) + 1;
}
print_r($by);
foreach ($items as $i) {
    $p = json_decode($i->payload, true);
    echo "--- #{$i->id} {$i->qtype} [{$i->status}] issues=" . json_encode($p['issues'] ?? []) . "\n";
    echo mb_substr(json_encode($p, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES), 0, 420) . "\n";
}
// Regenerate the first multichoice with a comment.
$mc = null;
foreach ($items as $i) {
    if ($i->qtype === 'multichoice') {
        $mc = $i;
        break;
    }
}
if ($mc) {
    $ctx = \context_course::instance($course->id);
    $reg = new regenerator(client_factory::create($ctx, $admin->id));
    $new = $reg->regenerate($mc, $job, $source, 'Make it about the storm and easier.', $admin->id);
    echo "=== regenerated #{$new->id}: " . mb_substr($new->payload, 0, 420) . "\n";
}
// Approve all pending and publish with a quiz.
foreach ($DB->get_records('local_aiquestiongen_item', ['jobid' => $jobid]) as $i) {
    if ($i->status === 'pending') {
        $DB->set_field('local_aiquestiongen_item', 'status', 'approved', ['id' => $i->id]);
    }
}
$job = $DB->get_record('local_aiquestiongen_job', ['id' => $jobid]);
$res = publisher::publish($job, $source, $admin->id, 'new:' . \context_course::instance($course->id)->id, true, 'new');
echo "publish: " . json_encode($res) . "\n";
echo "questions in bank: " . $DB->count_records_select('question', 'id IN (SELECT questionid FROM {local_aiquestiongen_item} WHERE jobid = ? AND questionid IS NOT NULL)', [$jobid]) . "\n";
echo "COURSE {$course->id}\n";
